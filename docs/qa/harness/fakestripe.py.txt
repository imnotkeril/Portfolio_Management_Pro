import json, threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
SUBS = {}
CALLS = []
class H(BaseHTTPRequestHandler):
    def _send(self, code, obj):
        b = json.dumps(obj).encode()
        self.send_response(code); self.send_header("Content-Type", "application/json"); self.send_header("Content-Length", str(len(b))); self.end_headers(); self.wfile.write(b)
    def log_message(self, *a): pass
    def do_GET(self):
        p = self.path.split("?")[0]
        if p == "/_calls": return self._send(200, CALLS)
        if p.startswith("/v1/subscriptions/"):
            sid = p.rsplit("/", 1)[1]; CALLS.append(sid)
            if self.headers.get("Authorization", "").startswith("Bearer sk_test_") is False: return self._send(401, {"error": {"type": "authentication_error", "message": "bad key"}})
            if sid.startswith("sub_boom"): return self._send(500, {"error": {"type": "api_error", "message": "boom"}})
            if sid not in SUBS: return self._send(404, {"error": {"type": "invalid_request_error", "code": "resource_missing", "message": "No such subscription: '%s'" % sid, "param": "subscription"}})
            return self._send(200, SUBS[sid])
        self._send(404, {"error": {"type": "invalid_request_error", "message": "unknown " + p}})
    def do_POST(self):
        n = int(self.headers.get("Content-Length", 0)); body = json.loads(self.rfile.read(n) or b"{}")
        if self.path == "/_set":
            SUBS[body["id"]] = body; return self._send(200, {"ok": True})
        if self.path == "/_del":
            SUBS.pop(body["id"], None); return self._send(200, {"ok": True})
        self._send(404, {})
ThreadingHTTPServer(("0.0.0.0", 12111), H).serve_forever()
