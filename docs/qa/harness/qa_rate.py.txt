def res(i, desc, ok, detail):
    OUT.append(dict(id=i, desc=desc, ok=bool(ok), detail=str(detail)))
    print(("PASS " if ok else "FAIL "), i, desc, "|", detail, flush=True)


def signed(ip, age=0, key=None):
    ts = str(int(time.time()) - age)
    sig = hmac.new((key or sec["CIP"]).encode(), f"{ip}|{ts}".encode(), hashlib.sha256).hexdigest()
    return {"X-Client-IP": ip, "X-Client-IP-Timestamp": ts, "X-Client-IP-Signature": sig}


def bad_login(base, email, hdr=None, path="/auth/login/json", pw="wrongwrong1"):
    if path.endswith("json"):
        return requests.post(base + path, json={"email": email, "password": pw}, headers=hdr or {})
    return requests.post(base + path, data={"username": email, "password": pw}, headers=hdr or {})


def mkuser(base):
    e = f"rl{uuid.uuid4().hex[:8]}@example.com"
    r = requests.post(base + "/auth/register", json={"email": e, "password": PW}); assert r.status_code == 201, r.text
    return e


def wait(n=62):
    print(f"   ...waiting {n}s for rate windows", flush=True); time.sleep(n)


def trip(codes, n=20):
    return codes[:n] == [401] * n and set(codes[n:]) == {429} and len(codes) > n

def bff_login(email, ip=None, extra=None):
    h = dict(extra or {})
    if ip: h["X-Real-IP"] = ip
    return requests.post(W + "/api/auth/login", json={"email": email, "password": "wrongwrong1"}, headers=h, allow_redirects=False)
