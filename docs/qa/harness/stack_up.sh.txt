#!/bin/bash
# Throwaway QA stack. No real keys: everything generated here.
cd ${QA_DIR:?set QA_DIR to a scratch dir containing fakestripe.py and shim/sitecustomize.py}
S=${QA_DIR:?set QA_DIR to a scratch dir containing fakestripe.py and shim/sitecustomize.py}
export MSYS_NO_PATHCONV=1
python - <<'PYEOF' > ports.env
import random,socket
def free(used):
    while True:
        p=random.randint(20000,39980)
        if any(abs(p-u)<6 for u in used): continue
        s=socket.socket()
        try: s.bind(("127.0.0.1",p)); s.close(); return p
        except OSError: pass
u=[]
for n in ("API_PORT","API2_PORT","API3_PORT","WEB_PORT"):
    p=free(u); u.append(p); print(f"{n}={p}")
PYEOF
source ports.env
JWT=$(python -c "import secrets;print(secrets.token_urlsafe(64))"); WHSEC=whsec_$(python -c "import secrets;print(secrets.token_hex(20))"); SK=sk_test_$(python -c "import secrets;print(secrets.token_hex(16))"); CIP=$(python -c "import secrets;print(secrets.token_urlsafe(48))")
printf "JWT=%s\nWHSEC=%s\nSK=%s\nCIP=%s\n" "$JWT" "$WHSEC" "$SK" "$CIP" > secrets.env
PGPW=$(python -c "import secrets;print(secrets.token_hex(12))")
DBURL="postgresql+psycopg2://qa:$PGPW@audit-pmp-qa-pg:5432/pmp_qa_test"
docker network create audit-pmp-qa-net >/dev/null
docker run -d --name audit-pmp-qa-pg --network audit-pmp-qa-net -e POSTGRES_USER=qa -e POSTGRES_PASSWORD=$PGPW -e POSTGRES_DB=pmp_qa_test postgres:16-alpine >/dev/null
docker run -d --name audit-pmp-qa-fakestripe --network audit-pmp-qa-net -p 127.0.0.1:$((API_PORT+1)):12111 -v "$S/fakestripe.py:/f.py:ro" --entrypoint python audit-pmp-qa-img /f.py >/dev/null
sleep 8
run() { N=$1; PORT=$2; TPC=$3; KEY=$4; docker run -d --name audit-pmp-qa-$N --network audit-pmp-qa-net -p 127.0.0.1:$PORT:8000 \
  -v "$S/shim:/shim:ro" -e PYTHONPATH=/shim -e FAKE_STRIPE_BASE=http://audit-pmp-qa-fakestripe:12111 \
  -e ENVIRONMENT=production -e JWT_SECRET="$JWT" -e DATABASE_URL="$DBURL" -e STRIPE_SECRET_KEY="$SK" -e STRIPE_WEBHOOK_SECRET="$WHSEC" -e TRUSTED_PROXY_COUNT=$TPC \
  ${KEY:+-e CLIENT_IP_SIGNING_KEY="$KEY"} -e FRONTEND_URL=http://127.0.0.1:$WEB_PORT audit-pmp-qa-img >/dev/null; }
run api $API_PORT 0 "$CIP"; sleep 14
run api-xff $((API_PORT+2)) 1 ""
run api-bff $((API_PORT+3)) 0 "$CIP"
docker run -d --name audit-pmp-qa-web --network audit-pmp-qa-net -p 127.0.0.1:$WEB_PORT:3000 -e API_INTERNAL_URL=http://audit-pmp-qa-api-bff:8000 -e CLIENT_IP_SIGNING_KEY="$CIP" -e CLIENT_IP_TRUSTED_HEADER=x-real-ip audit-pmp-qa-web >/dev/null
sleep 20
for p in 0 2 3; do curl -s localhost:$((API_PORT+p))/health; done; curl -s -o /dev/null -w " web=%{http_code}\n" localhost:$WEB_PORT/login
docker ps --format '{{.Names}} {{.Status}}' | grep audit-pmp-qa
