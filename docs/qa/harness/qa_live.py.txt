def sign(payload: bytes, secret=WHSEC, t=None):
    t = int(t if t is not None else time.time())
    sig = hmac.new(secret.encode(), f"{t}.".encode() + payload, hashlib.sha256).hexdigest()
    return f"t={t},v1={sig}"


def event(typ, obj, eid=None, created=None):
    return {"id": eid or "evt_" + uuid.uuid4().hex[:24], "object": "event", "api_version": "2024-06-20",
            "created": int(created or time.time()), "type": typ, "livemode": False, "data": {"object": obj}}


def hook(ev, secret=WHSEC, t=None, header=True, raw=None, sig_override=None):
    payload = raw if raw is not None else json.dumps(ev).encode()
    hd = {"Content-Type": "application/json"}
    if header: hd["Stripe-Signature"] = sig_override or sign(payload, secret, t)
    return requests.post(A + "/stripe/webhook", data=payload, headers=hd)
